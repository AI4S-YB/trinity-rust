```bash
#!/bin/bash
# sweep.sh <reads_prefix> <ref> <engine> —— 网格: max_cov{off,50,100,200} × K{19,23,25,29}
set -e
PRE=$1; REF=$2; ENGINE=$3   # e.g. dev1sub ../refs/dev1.fa recursive
cd "$(dirname "$0")/.."
ROOT=${ATHAL:-/storage/home/senior007/test/athal}/bench
echo "| max_cov | K | R | P | F1 | N50 | n |"
echo "|---|---|---|---|---|---|---|"
for cov in off 50 100 200; do
  for k in 19 23 25 29; do
    EXTRA="--KMER_SIZE $k"
    [ "$cov" != off ] && EXTRA="$EXTRA --normalize_max_read_cov $cov" || EXTRA="$EXTRA --no_normalize_reads"
    [ "$ENGINE" = recursive ] && EXTRA="$EXTRA --recursive"
    OUT=$(cargo run -q --release -p xtask -- bench --tag sw-${ENGINE}-c${cov}-k${k} \
      --reads $ROOT/reads/${PRE}_1.fq.gz,$ROOT/reads/${PRE}_2.fq.gz --ref $ROOT/refs/$REF.fa \
      -- $EXTRA --max_memory 8G 2>&1 | grep '^  转录本')
    echo "| $cov | $k | $OUT |" | sed 's/  转录本 //; s/ R=/ | /; s/ R70=[0-9.]*%//; s/ P=/ | /; s/ F1=/ | /; s/% | N50=/ | /; s/ N50=/ | /'
  done
done
```

- [ ] **Step 3: 实现 run_mini**（签名固定如下）

```rust
pub struct MiniParams { pub k: usize, pub strand: bool, pub min_contig_length: usize,
                        pub path_reinforcement_distance: usize, pub max_pair_distance: usize }

pub struct MiniOut { pub fasta: Vec<u8>, pub n_partitions_run: usize }

/// 单分区迷你 Trinity（TRINITY_COMPLETE 语义）:
/// reads_tmp 字节（">name/1 pct\nseq" 格式, /1 /2 保留）→ inchworm(内置kmer) → min100 → chrysalis 全链 → butterfly → fasta 片段
pub fn run_mini(reads_tmp: &[u8], p: &MiniParams) -> Result<MiniOut, CommonError> {
    let tmp = std::env::temp_dir().join(format!("mini_{}", std::process::id()));
    let _ = std::fs::remove_dir_all(&tmp); std::fs::create_dir_all(&tmp)?;
    let r = (|| -> Result<MiniOut, CommonError> {
        // ① inchworm from reads（TRINITY_COMPLETE: --reads 装载; 剪枝开关按"原版 phase-2 语义"节的核实结论）
        let mut counter = trinity_inchworm::irke::populate_from_reads(reads_tmp, p.k, !p.strand)?.0;
        trinity_inchworm::irke::prune_some_kmers(&mut counter, 1, /*min_any_entropy 按核实*/0.0,
                                                 /*prune_error 按核实*/false, 0.005);
        // ② 单线程排序种子组装（NO_PARALLEL_IWORM）
        let mut buf = Vec::new();
        trinity_inchworm::irke::compute_sequence_assemblies(&mut counter, &IrkeParams::default(),
            &AssemblyParams { min_assembly_length: p.k, ..Default::default() }, &Monitor::new(0),
            &mut buf)?;   // ← 签名以 irke.rs 实际为准（读后再写; 单线程入口在 P2 已有）
        // ③ min100
        let iworm = crate::orchestrate::filter_iworm_min_len_or_cov(&buf, 100, 10)?;
        // ④ chrysalis 全链（pct 50 钳制同主线）
        let listing = trinity_chrysalis::pipeline::run_chrysalis_pipeline(&iworm, reads_tmp,
            &tmp.join("chry"), &ChrysalisParams { strand: p.strand, min_contig_length: p.min_contig_length,
            pct_required: 50, threads: 1, ..Default::default() })?;
        // ⑤ butterfly 每组件（run_component 自带大栈线程）
        let mut fasta = Vec::new();
        for (_id, base) in &listing {
            let g = std::fs::read(base.with_extension("graph.out").clone().into_boxed_path())?;
            let r = std::fs::read(base.with_extension("graph.reads"))?;
            let out = trinity_butterfly::component::run_component(&g, &r, &ComponentParams {
                min_len: p.min_contig_length, max_pair_distance: p.max_pair_distance,
                path_reinforcement_distance: Some(p.path_reinforcement_distance),
                no_em_reduce: true, ..Default::default() })?;
            fasta.extend_from_slice(out.all_prob_paths_fasta.as_bytes());
        }
        Ok(MiniOut { fasta, n_partitions_run: 1 })
    })();
    let _ = std::fs::remove_dir_all(&tmp);
    r
}
```
