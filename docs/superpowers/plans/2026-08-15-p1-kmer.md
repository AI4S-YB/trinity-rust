```rust
//! 并行 k-mer 计数 — jellyfish count 的 Rust 替代（多重集等价，见 xcheck-kmer）
//! 语义: 小写 gatc 视同大写; 窗口遇非 gatc 跳过该 kmer 并继续滑 1 碱基;
//! DS 模式表键 = max(kmer, revcomp(kmer))（sequenceUtil.cpp:376）。

use rayon::prelude::*;
use rustc_hash::FxHashMap;

use trinity_common::kmer::{get_ds_kmer_val, revcomp_val, KmerId};

pub type CountMap = FxHashMap<KmerId, u32>;

/// 对一条序列（原始字节）枚举有效 kmer，回调规范化后的键。
/// 镜像 jellyfish: 非 gatc 字符使包含它的窗口失效，窗口继续前滑。
pub fn for_each_kmer(seq: &[u8], k: usize, ds: bool, mut f: impl FnMut(KmerId)) {
    let mask: KmerId = if k >= 32 { u64::MAX } else { (1u64 << (2 * k as u32)) - 1 };
    let mut kmer: KmerId = 0;
    let mut valid: usize = 0;
    for &c in seq {
        let v = match c {
            b'G' | b'g' => 0u64,
            b'A' | b'a' => 1,
            b'T' | b't' => 2,
            b'C' | b'c' => 3,
            _ => {
                valid = 0;
                kmer = 0;
                continue;
            }
        };
        kmer = ((kmer << 2) | v) & mask;
        if valid < k {
            valid += 1;
        }
        if valid == k {
            let key = if ds { get_ds_kmer_val(kmer, k) } else { kmer };
            f(key);
        }
    }
}

pub fn estimate_hash_size(max_memory: u64, read_file_size: u64) -> u64 {
    let mut h = if max_memory > read_file_size {
        (max_memory - read_file_size) / 7
    } else {

    if h < 100_000_000 || read_file_size < 5_000_000_000 {
```

```rust
impl KmerCountTable {
    /// 并行统计 FASTA 字节流的 k-mer 计数。
    /// rayon 按记录批分片 → 每线程局部 FxHashMap → reduce 合并（jellyfish 同型策略）。
    pub fn count_fasta_data(data: &[u8], k: usize, ds: bool) -> CountMap {
        let seqs: Vec<&[u8]> = split_fasta_sequences(data);
        seqs.par_iter()
            .fold(
                || -> CountMap { FxHashMap::default() },
                |mut local, seq| {
                    for_each_kmer(seq, k, ds, |key| *local.entry(key).or_insert(0) += 1);
                    local
                },
            )
            .reduce(
                || -> CountMap { FxHashMap::default() },
                |mut a, b| {
                    for (key, c) in b {
                        *a.entry(key).or_insert(0) += c;
                    }
                    a
                },
            )
    }
}

/// 切出 FASTA 的每条序列字节（跳过 header 行；不去空白/不大写——for_each_kmer 容忍）
fn split_fasta_sequences(data: &[u8]) -> Vec<&[u8]> {
    let mut out = Vec::new();
    let mut start: Option<usize> = None;
    let mut i = 0;
    while i < data.len() {
        let line_end = data[i..].iter().position(|&b| b == b'\n').map(|p| i + p).unwrap_or(data.len());
        let line = &data[i..line_end];
        if line.first() == Some(&b'>') {
            if let Some(s) = start.take() {
                out.push(&data[s..prev_nonnewline_end(i, data)]);
            }
            start = None; // header 之后序列从下一行开始
            let _ = start;
            // 序列起点 = header 行结束后
            start = if line_end < data.len() { Some(line_end + 1) } else { None };
        }
        i = line_end + 1;
    }
    if let Some(s) = start {
        out.push(&data[s..data.len()]);
    }
    out
}

fn prev_nonnewline_end(pos: usize, data: &[u8]) -> usize {
    let mut e = pos;
    while e > 0 && (data[e - 1] == b'\n' || data[e - 1] == b'\r') {
        e -= 1;
    }
    e
}
```

```rust
pub fn write_dump<W: Write>(w: &mut W, counts: &CountMap, k: usize, min_count: u32) -> std::io::Result<()> {

        if c < min_count {

        writeln!(w, ">{c}")?;
```

```rust
//! trinity-kmer CLI — jellyfish count+dump 的一体化替代 + 后续子命令。
//!
//! count:  trinity-kmer count --reads in.fa -K 25 [--canonical] [--min-count 1] [-o out.fa]

use std::io::BufReader;

use trinity_common::io_util::open_maybe_gz;

fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    let cmd = args.first().cloned().unwrap_or_default();
    let rest = &args[1..];
    let code = match cmd.as_str() {
        "count" => cmd_count(rest),
        _ => {
            eprintln!("用法: trinity-kmer <count|coverage-stats|diginorm> [参数]");
            2
        }
    };
    std::process::exit(code);
}

fn arg(rest: &[String], name: &str) -> Option<String> {
    rest.iter().position(|a| a == name).and_then(|i| rest.get(i + 1)).cloned()
}

fn cmd_count(rest: &[String]) -> i32 {
    let Some(reads) = arg(rest, "--reads") else {
        eprintln!("Error, --reads required");
        return 2;
    };
    let k: usize = arg(rest, "-K").and_then(|v| v.parse().ok()).unwrap_or(25);
    let canonical = rest.iter().any(|a| a == "--canonical");
    let min_count: u32 = arg(rest, "--min-count").and_then(|v| v.parse().ok()).unwrap_or(1);
    // gzip 魔数嗅探读入（.fa 与 .fa.gz 透明）
    let mut data = Vec::new();
    let read_res = match open_maybe_gz(std::path::Path::new(&reads)) {
        Ok(mut r) => r.read_to_end(&mut data),
        Err(e) => Err(std::io::Error::other(e)),
    };
    if let Err(e) = read_res {
        eprintln!("Error, cannot read {reads}: {e}");
        return 1;
    }
    let counts = trinity_kmer::counter::KmerCountTable::count_fasta_data(&data, k, canonical);
    let stdout = std::io::stdout();
    let mut w = BufWriter::new(stdout.lock());
    if let Err(e) = trinity_kmer::dump::write_dump(&mut w, &counts, k, min_count) {
        eprintln!("Error writing dump: {e}");
        return 1;
    }
    0
}
```

```rust
//! C++ `std::ostream <<` 默认浮点格式（≈ printf %g，6 位有效数字，去尾零）。
//! fastaToKmerCoverageStats 的 stats 输出依赖此格式（含 "nan"/"-0" 特例）。

pub fn format_g6(v: f32) -> String {
    if v.is_nan() {
        return if v.is_sign_negative() { "-nan".into() } else { "nan".into() };
    }
    if v.is_infinite() {
        return if v < 0.0 { "-inf".into() } else { "inf".into() };
    }
    if v == 0.0 {
        return if v.is_sign_negative() { "-0".into() } else { "0".into() };
    }
    let neg = v < 0.0;
    let a = v.abs() as f64;
    let p = 6usize;
    // 先以科学计数取 p 位有效数字
    let sci = format!("{:.*e}", p - 1, a);
    // 解析 "d.ddddde±X"
    let (mant_str, exp_str) = sci.split_once('e').unwrap();
    let exp: i32 = exp_str.parse().unwrap();
    let mant: f64 = mant_str.parse().unwrap();
    // %g 规则: 指数 < -4 或 >= p 用科学计数，否则定点
    let mut body = if exp < -4 || exp >= p as i32 {
        let mut m = format!("{:.p$e}", mant, p = p - 1);
        // 去尾零
        if m.contains('.') {
            while m.ends_with('0') { m.pop(); }
            if m.ends_with('.') { m.pop(); }
        }
        format!("{m}e{exp:+03}", exp = exp)
    } else {
        let decimals = (p as i32 - 1 - exp).max(0) as usize;
        let mut m = format!("{:.*}", decimals, mant);
        if m.contains('.') {
            while m.ends_with('0') { m.pop(); }
            if m.ends_with('.') { m.pop(); }
        }
        m
    };
    if body.contains('e') {
        // C 指数至少 2 位: e+05 / e-05（glibc）
        // （上面的 {exp:+03} 已产 +05 形态；断言留测试）
    }
    if neg {
        body.insert(0, '-');
    }
    body
}
```

```rust
pub fn median_cov(vals: &mut Vec<u32>) -> u32 {
    if vals.is_empty() { return 0; }
    vals.sort_unstable();
    let n = vals.len();
    if n % 2 == 1 { vals[n / 2] } else { (vals[(n - 1) / 2] + vals[n / 2]) / 2 } // 整数截断
}

pub fn mean_f32(vals: &[u32]) -> f32 {
    if vals.is_empty() { return 0.0; }
    let sum: i64 = vals.iter().map(|&v| v as i64).sum();
    sum as f32 / vals.len() as f32
}

pub fn stdev_f32(vals: &[u32]) -> f32 {
    let n = vals.len();
    let avg = mean_f32(vals);
    let mut sum: f32 = 0.0;
    for &v in vals {
        let d = v as f32 - avg;
        sum += d * d;
    }
    // n=0: 0/(0-1) = -0 → sqrt(-0) = -0; n=1: 0/0 = NaN（f32 除法语义与 C++ 一致）
    (sum / ((n as i64 - 1) as f32)).sqrt()
}
```

```rust
//! Perl_drand48 兼容 PRNG —— perl >= 5.20 的 rand() 即经典 drand48。
//! 目的: 复刻 nbkc_normalize.pl 的 srand(12345)+rand(1) 序列，使选择名单与原版逐字节一致。
//! 算法: 48-bit LCG, a=0x5DEECE66D, c=0xB; srand48(seed): X = (seed<<16)|0x330E (mod 2^48);
//! drand48() = next X / 2^48。

pub struct Drand48 { state: u64 } // 仅低 48 位有效

impl Drand48 {
    pub fn new(seed: u64) -> Self {
        Drand48 { state: ((seed << 16) | 0x330E) & 0xFFFF_FFFF_FFFF }
    }
    pub fn next_f64(&mut self) -> f64 {
        self.state = (self.state.wrapping_mul(0x5DEECE66D).wrapping_add(0xB)) & 0xFFFF_FFFF_FFFF;
        self.state as f64 / (1u64 << 48) as f64
    }
}
```

```rust
pub struct NbkcParams { pub max_cov: f64, pub min_cov: f64, pub max_cv: f64 }

pub struct StatsRow { pub acc: String, pub median: f64, pub mean: f64, pub stdev: f64 }

/// 返回保留的 core acc（顺序 = 输入顺序，镜像原版打印序）
pub fn select(rows: &[StatsRow], p: &NbkcParams, rng: &mut Drand48) -> Vec<String> {
    let mut out = Vec::new();
    for r in rows {
        if r.median < p.min_cov { continue; }
        if r.mean <= 0.0 { continue; }
        let cv = r.stdev / r.mean;
        if cv > p.max_cv { continue; }
        // NaN cv 比较为 false → 通过（镜像 Perl NaN 语义）
        if rng.next_f64() <= p.max_cov / r.median {
            out.push(core_acc(&r.acc));
        }
    }
    out
}

pub fn core_acc(acc: &str) -> String {
    acc.strip_suffix("/1").or_else(|| acc.strip_suffix("/2")).unwrap_or(acc).to_string()
}
```

```rust
//! seqtk-trinity 的 update_read_name_for_Trinity 移植（seqtk.c:200-283）+ fq→fa 转换。
//! 名字规则（原序）:
//!   1. 名字含 "_forward"/"_reverse" → 截断到该处
//!   2. 名字以 "/1"|"/2" 结尾 → 校验与期望类型一致（不一致 = 原版 exit(2)），原样返回
//!   3. comment 以 "1:"|"2:" 开头（Illumina 新式）→ 校验后追加 "/"+comment[0]
//!   4. 兜底 → 追加 "/1" 或 "/2"（原版 2014 年后不再报错）

use trinity_common::error::CommonError;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ReadType { R1 = 1, R2 = 2 }

pub fn update_read_name_for_trinity(
    name: &str,
    comment: Option<&str>,
    read_type: ReadType,
) -> Result<String, CommonError> {
    let mut name = name.to_string();
    // 1. _forward/_reverse 截断（注意: 原版 read_type_char 在截断前取，但截断后不参与 /1|/2 分支）
    let mut truncated = false;
    if let Some(pos) = name.find("_forward").or_else(|| name.find("_reverse")) {
        name.truncate(pos);
        truncated = true;
    }
    let name_len = name.len();
    if !truncated && name_len >= 2 {
        let bytes = name.as_bytes();
        let last = bytes[name_len - 1];
        if bytes[name_len - 2] == b'/' && (last == b'1' || last == b'2') {
            let found_type = if last == b'1' { ReadType::R1 } else { ReadType::R2 };
            if found_type != read_type {
                return Err(CommonError::FastaFormat(format!(
                    "Error, found read_type {} but expecting read_type {}",
                    last as char, read_type as u8
                )));
            }
            return Ok(name);
        }
    }
    // 3. Illumina 新式 comment
    if !truncated {
        if let Some(c) = comment {
            let cb = c.as_bytes();
            if cb.len() > 1 && cb[1] == b':' && (cb[0] == b'1' || cb[0] == b'2') {
                let found_type = if cb[0] == b'1' { ReadType::R1 } else { ReadType::R2 };
                if found_type != read_type {
                    return Err(CommonError::FastaFormat(format!(
                        "Error, found read_type {} but expecting read_type {}",
                        cb[0] as char, read_type as u8
                    )));
                }
                return Ok(format!("{name}/{}", cb[0] as char));
            }
        }
    }
    // 4. 兜底
    Ok(format!("{name}/{}", read_type as u8 as char))
}
```
⚠️ 原版细节：`_forward` 截断后走兜底分支**追加 /1|/2**（截断名 + "/1"）——上面 `truncated` 标志实现了这一点（原版 `found != NULL` 时跳过分支 2 和 3）。

```rust
//! DigiNorm 编排 — insilico_read_normalization.pl 的 together 模式镜像:
//! fq→fa（名字规范 + SS 'R' 端 revcomp）→ 计数（dump -L 2）→ 双端 stats → merge → nbkc 选择
//! → 按名单扫描原始文件抽取记录（原格式原样写出）。

use std::path::{Path, PathBuf};

use trinity_common::error::CommonError;

use crate::counter::{self, KmerCountTable};
use crate::drand48::Drand48;
use crate::nbkc::{self, NbkcParams};
use crate::read_names::ReadType;

pub struct DigiNormParams {
    pub k: usize,             // 25
    pub min_kmer_cov_const: u32, // 2（dump -L，硬编码"DO NOT CHANGE"）
    pub max_cov: f64,         // 200
    pub min_cov: f64,         // 1
    pub max_cv: f64,          // 10000
    pub ss_revcomp_left: bool,   // SS RF: left 端 revcomp
    pub ss_revcomp_right: bool,  // SS（一般 FR 不 revcomp；R 单端 revcomp）
}

impl Default for DigiNormParams {
    fn default() -> Self {
        DigiNormParams { k: 25, min_kmer_cov_const: 2, max_cov: 200.0, min_cov: 1.0, max_cv: 10000.0,
                         ss_revcomp_left: false, ss_revcomp_right: false }
    }
}

pub struct DigiNormOutputs { pub left: Option<PathBuf>, pub right: Option<PathBuf>, pub single: Option<PathBuf> }

/// single 或 PE-together。PE: kmer 表来自两端合并的 fa（原版 both.fa）。
pub enum ReadsInput { Single(PathBuf), Paired(PathBuf, PathBuf) }

pub fn run(
    params: &DigiNormParams,
    reads: &ReadsInput,
    out_dir: &Path,
) -> Result<DigiNormOutputs, CommonError>
```

步骤实现（每步一个私有函数，顺序执行）：
1. `prep`: fq/fa 读入 →（fq 时）fq_records_to_fa（-R 1/-R 2 + revcomp 标志）；fa 输入直接用（名字假定已规范——原版对 fa 输入也过 prep？执行者读 insilico_read_normalization.pl:660-719 确认：fa 也走 seqtk？若原版 fa 直接用，我们对 fa 直接用）。PE 合并 both.fa
2. `count`: `KmerCountTable::count_fasta_data(both_or_single, k, ds=!ss)` → 内存 dump（不落盘；`--no_cleanup` 场景 P5 再说）→ `counts`
3. `stats`: 双端（或单端）各自 `coverage_stats_rows`（从 counts 直接查，省去 dump 文件往返）；rows 按 acc 字节序排序（镜像原版 stats.sort）
4. PE: `nbkc::merge_pairs` → rows（median/mean/stdev 从 %.1f 字符串**再解析回 f64**——镜像原版管道：merge 输出字符串、nbkc 按列名再解析）
5. `select`: `Drand48::new(12345)` + `NbkcParams` → core acc 名单
6. `extract`: 扫描**原始**输入文件（fq 逐 4 行 / fa 逐记录，acc → core 名匹配：strip `/1|/2`、`_forward|_reverse` 尾巴——镜像 PerlLib/Fastq_reader.pm get_core_read_name 的 `s/_forward//; s/_reverse//`），命中即原样写出记录（含原始 header/质量行）
7. 输出命名（镜像 L398-410）: `{base}.normalized_K{K}_maxC{max}_minC{min}_maxCV{cv}.fa|.fq`；out_dir 下同时写 `left.norm.fq`/`right.norm.fq`/`single.norm.fq`（按输入类型）——直接以 norm 名写出即可（便捷名与长名都写，镜像原版软链行为改为双写）
